# Astro Runtime 3.0-14 ships Apache Airflow 3.0.6 pre-installed.
# We add: Snowflake provider, dbt + adapter, AWS SDK, HTTP client, retry helper.
#
# All deps pinned EXACTLY — loose ranges trigger pip backtracking hell on
# dbt-adapters resolution (every dbt-snowflake patch combinatorially explodes
# against every dbt-adapters version). Lockstep upgrade in one PR if needed.

# Provider versions compatible with Airflow 3.x. snowflake 6.x and amazon
# 9.5+ are the first lines that declare 3.x compatibility (older ones target
# 2.10 max).
apache-airflow-providers-snowflake==6.12.2
apache-airflow-providers-amazon==9.27.0

# typer 0.18+ removed the [all] extra. dbt's deps still request typer[all],
# which sends pip into multi-minute backtracking through every typer/dbt-*
# version combination. Pinning typer to the last version with [all] cuts
# the resolver dead.
typer==0.12.5

# conflicts with dbt 1.8's pin (`>=0.5.1,<0.6`). dbt 1.9 uses the newer
# interfaces and resolves cleanly. All features we use (constraints,
# custom tests, snapshots with strategy='check', sources/freshness,
# dbt_utils, dbt_expectations, Iceberg materialisation if needed later)
# work in 1.9.
dbt-core==1.9.10
dbt-snowflake==1.9.4
